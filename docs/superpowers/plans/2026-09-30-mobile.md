# Telefon i podsetnici — plan implementacije

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Član na iPhone-u doda unos za manje od minuta, vidi isti mesec kao na webu, i svi članovi dobiju jedan push za podsetnik i za prag limita.

**Architecture:** Expo Router aplikacija koristi isti Supabase projekat i `@finance/domain`. Zakazani posao u `apps/jobs` jednom dnevno, u zoni Beograda, ubaci ponavljanje i pošalje push. Prag limita računa isti posao i ruta koju klijent pozove posle čuvanja. Duplikat sprečava `sent_notifications` i unique indeks pravilo+mesec.

**Tech Stack:** Expo, expo-router, expo-notifications, expo-apple-authentication, `@supabase/supabase-js`, `@finance/domain`, Vitest za posao.

## Global Constraints

- Prijava: link na e-poštu, plus Apple nalog na iPhone-u kad pošalje istu e-poštu.
- Ime za prikaz je obavezno.
- Član dodaje unos. Ne vidi izmene, brisanje, kategorije, limite ni pozivnice.
- Bez mreže forma ostaje popunjena i piše da unos nije sačuvan. Nema reda za naknadnu sinhronizaciju.
- Push stiže svim članovima. Na webu se isto vidi kao stavka u pregledu; ovaj plan šalje samo iPhone push.
- Podsetnik je 1–7 dana pre dospeća, podrazumevano 1. Za isto dospeće odlazi jednom.
- Dan 31 u kraćem mesecu je poslednji dan, preko `occurrenceDate`.
- Par pravilo i mesec daje tačno jedan unos.
- Prag 80 i prag 100 šalju se jednom po kategoriji i mesecu, preko `limitThresholds`.
- Vremenska zona je Europe/Belgrade, preko `todayInBelgrade`.
- Ručni unos pored automatskog ostaje poseban red.

`apps/jobs/src/decide.ts`

```ts
import { addCalendarDays, monthKey, occurrenceDate } from "@finance/domain";

export function occurrenceThisMonth(today: string, dayOfMonth: number): string {
  const [year, month] = today.split("-").map(Number);
  return occurrenceDate(year, month, dayOfMonth);
}

export function dueActions(input: {
  today: string;
  dayOfMonth: number;
  remindDays: number;
  active: boolean;
  reminderSent: boolean;
  entryExists: boolean;
}): Array<"remind" | "insert"> {
  if (!input.active) return [];
  const occurrence = occurrenceThisMonth(input.today, input.dayOfMonth);
  const reminder = addCalendarDays(occurrence, -input.remindDays);
  const actions: Array<"remind" | "insert"> = [];
  if (!input.reminderSent && input.today >= reminder && input.today < occurrence) {
    actions.push("remind");
  }
  if (!input.entryExists && input.today >= occurrence && monthKey(input.today) === monthKey(occurrence)) {
    actions.push("insert");
  }
  return actions;
}
```

  user_id uuid not null references public.profiles (id) on delete cascade,
  token text not null,
  primary key (user_id, token)

alter table public.push_tokens enable row level security;

`apps/jobs/src/run-day.ts`

```ts
import { createClient, type SupabaseClient } from "@supabase/supabase-js";
import { limitThresholds, monthKey, summarizeMonth, todayInBelgrade, type CategorySnapshot, type EntrySnapshot } from "@finance/domain";
import { dueActions, occurrenceThisMonth } from "./decide";

function admin(): SupabaseClient {
  return createClient(process.env.SUPABASE_URL!, process.env.SUPABASE_SERVICE_ROLE_KEY!, {
    auth: { persistSession: false },
  });
}

async function pushToHousehold(client: SupabaseClient, householdId: string, body: string) {
  const members = await client.from("memberships").select("user_id").eq("household_id", householdId);
  const ids = (members.data ?? []).map((row) => row.user_id);
  const tokens = await client.from("push_tokens").select("token").in("user_id", ids);
  if (!tokens.data?.length) return;
  await fetch("https://exp.host/--/api/v2/push/send", {
    method: "POST",
    headers: { "content-type": "application/json" },
    body: JSON.stringify(tokens.data.map((row) => ({ to: row.token, body }))),
  });
}

async function claim(client: SupabaseClient, householdId: string, dedupeKey: string): Promise<boolean> {
  const inserted = await client.from("sent_notifications").insert({ household_id: householdId, dedupe_key: dedupeKey });
  return !inserted.error;
}

export async function sendLimitAlerts(client: SupabaseClient, householdId: string, month: string) {
  const [categories, entries] = await Promise.all([
    client.from("categories").select("id, name, kind, limit_minor").eq("household_id", householdId),
    client.from("entries").select("id, kind, amount_minor, category_id, person_id, person_name, occurred_on").eq("household_id", householdId).eq("month_key", month),
  ]);
  const categorySnapshots: CategorySnapshot[] = (categories.data ?? []).map((row) => ({
    id: row.id,
    name: row.name,
    kind: row.kind,
    limitMinor: row.limit_minor,
  }));
  const entrySnapshots: EntrySnapshot[] = (entries.data ?? []).map((row) => ({
    id: row.id,
    kind: row.kind,
    amountMinor: row.amount_minor,
    categoryId: row.category_id,
    personId: row.person_id,
    personName: row.person_name,
    occurredOn: row.occurred_on,
  }));
  const summary = summarizeMonth({ month, categories: categorySnapshots, entries: entrySnapshots });
  for (const category of summary.categories) {
    for (const threshold of limitThresholds(category.spentMinor, category.limitMinor)) {
      const claimed = await claim(client, householdId, `limit:${category.categoryId}:${month}:${threshold}`);
      if (claimed) await pushToHousehold(client, householdId, `${category.name} je prešla ${threshold}% limita.`);
    }
  }
}

export async function runDay(now: Date) {
  const client = admin();
  const today = todayInBelgrade(now);
  const rules = await client.from("recurring_rules").select("id, household_id, kind, amount_minor, category_id, person_id, note, day_of_month, remind_days, active");
  for (const rule of rules.data ?? []) {
    const occurrence = occurrenceThisMonth(today, rule.day_of_month);
    const month = monthKey(occurrence);
    const existing = await client.from("entries").select("id").eq("recurring_rule_id", rule.id).eq("month_key", month).maybeSingle();
    const sent = await client.from("sent_notifications").select("id").eq("household_id", rule.household_id).eq("dedupe_key", `reminder:${rule.id}:${month}`).maybeSingle();
    const actions = dueActions({
      today,
      dayOfMonth: rule.day_of_month,
      remindDays: rule.remind_days,
      active: rule.active,
      reminderSent: Boolean(sent.data),
      entryExists: Boolean(existing.data),
    });
    if (actions.includes("insert")) {
      await client.from("entries").insert({
        household_id: rule.household_id,
        kind: rule.kind,
        amount_minor: rule.amount_minor,
        category_id: rule.category_id,
        person_id: rule.person_id,
        occurred_on: occurrence,
        note: rule.note,
        recurring_rule_id: rule.id,
        created_by: rule.person_id,
      });
    }
    if (actions.includes("remind")) {
      const claimed = await claim(client, rule.household_id, `reminder:${rule.id}:${month}`);
      if (claimed) {
        const category = await client.from("categories").select("name").eq("id", rule.category_id).single();
        await pushToHousehold(client, rule.household_id, `Podsetnik: ${category.data?.name ?? "unos"} dospeva ${occurrence}.`);
      }
    }
    await sendLimitAlerts(client, rule.household_id, monthKey(today));
  }
}
```

`apps/web/app/api/alerts/route.ts`

```ts
import { NextResponse } from "next/server";
import { createClient } from "@supabase/supabase-js";
import { monthKey, todayInBelgrade } from "@finance/domain";
import { createServerSupabase } from "@/lib/supabase/server";
import { sendLimitAlerts } from "jobs/src/run-day";

export async function POST(request: Request) {
  const supabase = await createServerSupabase();
  const user = await supabase.auth.getUser();
  if (!user.data.user) return NextResponse.json({ error: "Prijava je obavezna" }, { status: 401 });
  const body = (await request.json()) as { householdId?: string };
  if (!body.householdId) return NextResponse.json({ error: "Nema domaćinstva" }, { status: 400 });
  const membership = await supabase.from("memberships").select("role").eq("household_id", body.householdId).eq("user_id", user.data.user.id).maybeSingle();
  if (!membership.data) return NextResponse.json({ error: "Zabranjeno" }, { status: 403 });
  const admin = createClient(process.env.SUPABASE_URL!, process.env.SUPABASE_SERVICE_ROLE_KEY!, {
    auth: { persistSession: false },
  });
  await sendLimitAlerts(admin, body.householdId, monthKey(todayInBelgrade(new Date())));
  return NextResponse.json({ ok: true });
}
```

import * as Notifications from "expo-notifications";
import { useEffect } from "react";
import { supabase } from "../lib/supabase";

export function usePushToken() {
  useEffect(() => {
    void (async () => {
      const { status } = await Notifications.requestPermissionsAsync();
      if (status !== "granted") return;
      const token = await Notifications.getExpoPushTokenAsync();
      const user = await supabase.auth.getUser();
      if (!user.data.user) return;
      await supabase.from("push_tokens").upsert({
        user_id: user.data.user.id,
        token: token.data,
      });
    })();
  }, []);
}
