# Web pregled — plan implementacije

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Porodica na webu vidi isti mesec kao na telefonu: prihod, trošak, ostatak, limite, filter po osobi, jednu rečenicu predloga, i vlasnik podešava kategorije, članove i ponavljanja.

**Architecture:** Next.js App Router zove Supabase sa sesijom korisnika. Zbirove i rečenicu računa `@finance/domain`, ne SQL. Član vidi forme za unos. Forme za kategorije, limite, pozivnice, uklanjanje člana i izmenu pravila vidi samo vlasnik.

**Tech Stack:** Next.js, React, `@supabase/ssr`, `@supabase/supabase-js`, `@finance/domain`, Vitest.

## Global Constraints

- Svi članovi vide sve unose, limite i predlog. Filter po osobi sužava prikaz.
- Član sme da doda unos. Izmenu, brisanje, kategorije, limite i pozivnice radi samo vlasnik.
- Vlasnik menja i gasi pravila ponavljanja: iznos, dan, podsetnik.
- Prijava je link na e-poštu. Ime za prikaz je obavezno.
- Iznos nula ili ispod nule se odbija. U bazu ide ceo broj u parama.
- Prikaz novca ide kroz `formatMoney` iz `@finance/domain`.
- Predlog ide kroz `suggest`. Zbir meseca ide kroz `summarizeMonth`.
- Mesec je kalendarski, zona Europe/Belgrade, preko `todayInBelgrade` i `monthKey`.
- Bez mreže forma ostaje popunjena i piše da unos nije sačuvan.
- Početne kategorije dolaze iz `create_household`. Limiti su prazni dok ih vlasnik ne unese.
- Valuta se bira pri otvaranju i posle se ne menja.
- Otvoren ekran dobija nove unose čim su sačuvani, preko Supabase realtime kanala `entries`.
- Na webu se prag limita i naredni podsetnik vide kao stavke u pregledu, ne kao push.

```ts
import { assertPositiveMinor, type CategorySnapshot, type EntrySnapshot } from "@finance/domain";

export function toCategorySnapshot(row: {
  id: string;
  name: string;
  kind: "expense" | "income";
  limit_minor: number | null;
}): CategorySnapshot {
  return { id: row.id, name: row.name, kind: row.kind, limitMinor: row.limit_minor };
}

export function toEntrySnapshot(row: {
  id: string;
  kind: "expense" | "income";
  amount_minor: number;
  category_id: string;
  person_id: string;
  person_name: string;
  occurred_on: string;
}): EntrySnapshot {
  return {
    id: row.id,
    kind: row.kind,
    amountMinor: row.amount_minor,
    categoryId: row.category_id,
    personId: row.person_id,
    personName: row.person_name,
    occurredOn: row.occurred_on,
  };
}

export function majorToMinor(raw: string): number {
  const cleaned = raw.trim().replace(/\./g, "").replace(",", ".");
  const major = Number(cleaned);
  if (!Number.isFinite(major)) {
    throw new Error("Iznos mora biti pozitivan ceo broj.");
  }
  const minor = Math.round(major * 100);
  assertPositiveMinor(minor);
  return minor;
}

export function canManage(role: string): boolean {
  return role === "owner";
}
```

```ts
import { createServerClient } from "@supabase/ssr";
import { cookies } from "next/headers";

export async function createServerSupabase() {
  const cookieStore = await cookies();
  return createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
    {
      cookies: {
        getAll() {
          return cookieStore.getAll();
        },
        setAll(cookiesToSet) {
          for (const cookie of cookiesToSet) {
            cookieStore.set(cookie.name, cookie.value, cookie.options);
          }
        },
      },
    },
  );
}
```

```ts
import {
  formatMoney,
  limitThresholds,
  suggest,
  summarizeMonth,
  type CategorySnapshot,
  type EntrySnapshot,
} from "@finance/domain";

export function buildMonthView(input: {
  currency: string;
  month: string;
  categories: CategorySnapshot[];
  entries: EntrySnapshot[];
  previous: EntrySnapshot[] | null;
  personId?: string;
  personName?: string;
}) {
  const visible = input.personId
    ? input.entries.filter((entry) => entry.personId === input.personId)
    : input.entries;
  const summary = summarizeMonth({
    month: input.month,
    categories: input.categories,
    entries: visible,
  });
  const sentence = suggest({
    currency: input.currency,
    categories: input.categories,
    current: input.entries,
    previous: input.previous,
    personId: input.personId,
    personName: input.personName,
  }).sentence;
  const bars = summary.categories
    .filter((category) => category.kind === "expense" && category.spentMinor > 0)
    .map((category) => {
      const ratio = category.limitMinor
        ? Math.min(100, Math.round((category.spentMinor * 100) / category.limitMinor))
        : 100;
      const limitLabel = category.limitMinor
        ? ` / ${formatMoney(category.limitMinor, input.currency)}`
        : "";
      return {
        name: category.name,
        width: ratio,
        label: `${formatMoney(category.spentMinor, input.currency)}${limitLabel}`,
      };
    });
  const alerts = summary.categories.flatMap((category) => {
    const hits = limitThresholds(category.spentMinor, category.limitMinor);
    return hits.map((hit) => `${category.name} je prešla ${hit}% limita.`);
  });
  return {
    income: formatMoney(summary.incomeMinor, input.currency),
    expense: formatMoney(summary.expenseMinor, input.currency),
    leftover: formatMoney(summary.leftoverMinor, input.currency),
    sentence,
    bars,
    alerts,
  };
}
```

```tsx
import { todayInBelgrade, monthKey } from "@finance/domain";
import { createServerSupabase } from "@/lib/supabase/server";
import { toCategorySnapshot, toEntrySnapshot } from "@/lib/rows";
import { buildMonthView } from "@/lib/month-view";
import { LiveEntries } from "./live";

export default async function MonthPage({
  params,
  searchParams,
}: {
  params: Promise<{ householdId: string }>;
  searchParams: Promise<{ month?: string; person?: string }>;
}) {
  const { householdId } = await params;
  const query = await searchParams;
  const supabase = await createServerSupabase();
  const month = query.month ?? monthKey(todayInBelgrade(new Date()));
  const [year, monthNumber] = month.split("-").map(Number);
  const previousMonth = monthNumber === 1 ? `${year - 1}-12` : `${year}-${String(monthNumber - 1).padStart(2, "0")}`;
  const [household, categories, entries, members] = await Promise.all([
    supabase.from("households").select("currency, name").eq("id", householdId).single(),
    supabase.from("categories").select("id, name, kind, limit_minor").eq("household_id", householdId).eq("archived", false),
    supabase.from("entries").select("id, kind, amount_minor, category_id, person_id, person_name, occurred_on").eq("household_id", householdId).in("month_key", [month, previousMonth]),
    supabase.from("memberships").select("user_id, profiles(display_name)").eq("household_id", householdId),
  ]);
  const categorySnapshots = (categories.data ?? []).map(toCategorySnapshot);
  const entryRows = entries.data ?? [];
  const current = entryRows.filter((row) => row.occurred_on.startsWith(month)).map(toEntrySnapshot);
  const previous = entryRows.filter((row) => row.occurred_on.startsWith(previousMonth)).map(toEntrySnapshot);
  const personName = (members.data ?? [])
    .map((row) => {
      const profile = Array.isArray(row.profiles) ? row.profiles[0] : row.profiles;
      return { id: row.user_id, name: profile?.display_name ?? "" };
    })
    .find((person) => person.id === query.person)?.name;
  const view = buildMonthView({
    currency: household.data?.currency ?? "RSD",
    month,
    categories: categorySnapshots,
    entries: current,
    previous,
    personId: query.person,
    personName,
  });
  return (
    <section>
      <h1>{household.data?.name}</h1>
      <nav>
        {(members.data ?? []).map((row) => {
          const profile = Array.isArray(row.profiles) ? row.profiles[0] : row.profiles;
          return (
            <a key={row.user_id} href={`/h/${householdId}?month=${month}&person=${row.user_id}`}>
              {profile?.display_name}
            </a>
          );
        })}
        <a href={`/h/${householdId}?month=${month}`}>Svi</a>
      </nav>
      <p>Prihod {view.income}</p>
      <p>Trošak {view.expense}</p>
      <p>Ostatak {view.leftover}</p>
      <p>{view.sentence}</p>
      <ul>
        {view.alerts.map((alert) => (
          <li key={alert}>{alert}</li>
        ))}
      </ul>
      {view.bars.map((bar) => (
        <div key={bar.name}>
          <span>{bar.name}</span>
          <div style={{ background: "#ddd", height: 12 }}>
            <div style={{ width: `${bar.width}%`, height: 12, background: "#1c1915" }} />
          </div>
          <span>{bar.label}</span>
        </div>
      ))}
      <LiveEntries householdId={householdId} />
      <p><a href={`/h/${householdId}/novi`}>Novi unos</a></p>
      <p><a href={`/h/${householdId}/podesavanja`}>Podešavanja</a></p>
    </section>
  );
}
```

```tsx
"use client";

import { useEffect } from "react";
import { useRouter } from "next/navigation";
import { createBrowserSupabase } from "@/lib/supabase/client";

export function LiveEntries({ householdId }: { householdId: string }) {
  const router = useRouter();
  useEffect(() => {
    const supabase = createBrowserSupabase();
    const channel = supabase
      .channel(`entries-${householdId}`)
      .on(
        "postgres_changes",
        { event: "*", schema: "public", table: "entries", filter: `household_id=eq.${householdId}` },
        () => router.refresh(),
      )
      .subscribe();
    return () => {
      void supabase.removeChannel(channel);
    };
  }, [householdId, router]);
  return null;
}
```

```tsx
"use client";

import { useState } from "react";
import { useRouter } from "next/navigation";
import { assertRemindDays } from "@finance/domain";
import { createBrowserSupabase } from "@/lib/supabase/client";
import { majorToMinor } from "@/lib/rows";

type Option = { id: string; name: string };

export function EntryForm({
  householdId,
  categories,
  people,
}: {
  householdId: string;
  categories: Array<Option & { kind: "expense" | "income" }>;
  people: Option[];
}) {
  const router = useRouter();
  const [kind, setKind] = useState<"expense" | "income">("expense");
  const [amount, setAmount] = useState("");
  const [categoryId, setCategoryId] = useState(categories.find((item) => item.kind === "expense")?.id ?? "");
  const [personId, setPersonId] = useState(people[0]?.id ?? "");
  const [occurredOn, setOccurredOn] = useState(new Date().toISOString().slice(0, 10));
  const [note, setNote] = useState("");
  const [repeat, setRepeat] = useState(false);
  const [remindDays, setRemindDays] = useState("1");
  const [error, setError] = useState("");

  async function onSubmit(event: React.FormEvent) {
    event.preventDefault();
    setError("");
    try {
      const amountMinor = majorToMinor(amount);
      const days = Number(remindDays);
      if (repeat) assertRemindDays(days);
      const supabase = createBrowserSupabase();
      let recurringRuleId: string | null = null;
      if (repeat) {
        const rule = await supabase.from("recurring_rules").insert({
          household_id: householdId,
          kind,
          amount_minor: amountMinor,
          category_id: categoryId,
          person_id: personId,
          note,
          day_of_month: Number(occurredOn.slice(8, 10)),
          remind_days: days,
        }).select("id").single();
        if (rule.error) throw rule.error;
        recurringRuleId = rule.data.id;
      }
      const inserted = await supabase.from("entries").insert({
        household_id: householdId,
        kind,
        amount_minor: amountMinor,
        category_id: categoryId,
        person_id: personId,
        occurred_on: occurredOn,
        note,
        recurring_rule_id: recurringRuleId,
      });
      if (inserted.error) throw inserted.error;
      router.push(`/h/${householdId}`);
      router.refresh();
    } catch (caught) {
      setError(caught instanceof Error ? caught.message : "Unos nije sačuvan.");
    }
  }

  const visibleCategories = categories.filter((category) => category.kind === kind);

  return (
    <form onSubmit={onSubmit}>
      <label>
        Vrsta
        <select value={kind} onChange={(event) => setKind(event.target.value as "expense" | "income")}>
          <option value="expense">Trošak</option>
          <option value="income">Prihod</option>
        </select>
      </label>
      <label>
        Iznos
        <input value={amount} onChange={(event) => setAmount(event.target.value)} required />
      </label>
      <label>
        Kategorija
        <select value={categoryId} onChange={(event) => setCategoryId(event.target.value)}>
          {visibleCategories.map((category) => (
            <option key={category.id} value={category.id}>{category.name}</option>
          ))}
        </select>
      </label>
      <label>
        Osoba
        <select value={personId} onChange={(event) => setPersonId(event.target.value)}>
          {people.map((person) => (
            <option key={person.id} value={person.id}>{person.name}</option>
          ))}
        </select>
      </label>
      <label>
        Datum
        <input type="date" value={occurredOn} onChange={(event) => setOccurredOn(event.target.value)} required />
      </label>
      <label>
        Beleška
        <input value={note} onChange={(event) => setNote(event.target.value)} />
      </label>
      <label>
        <input type="checkbox" checked={repeat} onChange={(event) => setRepeat(event.target.checked)} />
        Mesečno ponavljanje
      </label>
      {repeat ? (
        <label>
          Podsetnik, dana ranije
          <input value={remindDays} onChange={(event) => setRemindDays(event.target.value)} />
        </label>
      ) : null}
      <button type="submit">Sačuvaj</button>
      {error ? <p>Unos nije sačuvan. {error}</p> : null}
    </form>
  );
}
```

```ts
"use server";

import { revalidatePath } from "next/cache";
import { createServerSupabase } from "@/lib/supabase/server";
import { canManage, majorToMinor } from "@/lib/rows";

async function requireOwner(householdId: string) {
  const supabase = await createServerSupabase();
  const { data } = await supabase.auth.getUser();
  const membership = await supabase
    .from("memberships")
    .select("role")
    .eq("household_id", householdId)
    .eq("user_id", data.user?.id ?? "")
    .single();
  if (!canManage(membership.data?.role ?? "")) {
    throw new Error("Samo vlasnik menja podešavanja.");
  }
  return supabase;
}

export async function saveLimit(householdId: string, categoryId: string, raw: string) {
  const supabase = await requireOwner(householdId);
  const limit = raw.trim() === "" ? null : majorToMinor(raw);
  const result = await supabase.from("categories").update({ limit_minor: limit }).eq("id", categoryId);
  if (result.error) throw new Error(result.error.message);
  revalidatePath(`/h/${householdId}`);
}

export async function archiveCategory(householdId: string, categoryId: string) {
  const supabase = await requireOwner(householdId);
  const result = await supabase.from("categories").update({ archived: true }).eq("id", categoryId);
  if (result.error) throw new Error(result.error.message);
  revalidatePath(`/h/${householdId}`);
}

export async function inviteMember(householdId: string, email: string) {
  const supabase = await requireOwner(householdId);
  const result = await supabase.from("invitations").insert({ household_id: householdId, email });
  if (result.error) throw new Error(result.error.message);
}

export async function removeMember(householdId: string, userId: string) {
  const supabase = await requireOwner(householdId);
  const result = await supabase.from("memberships").delete().eq("household_id", householdId).eq("user_id", userId);
  if (result.error) throw new Error(result.error.message);
}

export async function saveRule(
  householdId: string,
  ruleId: string,
  input: { amount: string; day: number; remindDays: number; active: boolean },
) {
  const supabase = await requireOwner(householdId);
  const result = await supabase.from("recurring_rules").update({
    amount_minor: majorToMinor(input.amount),
    day_of_month: input.day,
    remind_days: input.remindDays,
    active: input.active,
  }).eq("id", ruleId);
  if (result.error) throw new Error(result.error.message);
}
```

```tsx
import { createServerSupabase } from "@/lib/supabase/server";

export default async function SettingsPage({ params }: { params: Promise<{ householdId: string }> }) {
```

```tsx
import { notFound } from "next/navigation";
import { createServerSupabase } from "@/lib/supabase/server";
import { canManage } from "@/lib/rows";
import { deleteEntry, updateEntry } from "../../podesavanja/actions";

export default async function EntryPage({ params }: { params: Promise<{ householdId: string; entryId: string }> }) {
  const { householdId, entryId } = await params;
  const supabase = await createServerSupabase();
  const user = await supabase.auth.getUser();
  const [membership, entry, categories, members] = await Promise.all([
    supabase.from("memberships").select("role").eq("household_id", householdId).eq("user_id", user.data.user?.id ?? "").single(),
    supabase.from("entries").select("id, amount_minor, category_id, person_id, occurred_on, note, person_name").eq("id", entryId).single(),
    supabase.from("categories").select("id, name").eq("household_id", householdId).eq("archived", false),
    supabase.from("memberships").select("user_id, profiles(display_name)").eq("household_id", householdId),
  ]);
  if (!entry.data) notFound();
  if (!canManage(membership.data?.role ?? "")) {
    return <p>Unos menja vlasnik. {entry.data.person_name}</p>;
  }
  return (
    <form action={async (formData) => {
      "use server";
      await updateEntry(householdId, entryId, {
        amount: String(formData.get("amount") ?? ""),
        categoryId: String(formData.get("category") ?? ""),
        personId: String(formData.get("person") ?? ""),
        occurredOn: String(formData.get("date") ?? ""),
        note: String(formData.get("note") ?? ""),
      });
    }}>
      <input name="amount" defaultValue={String((entry.data.amount_minor ?? 0) / 100)} />
      <select name="category" defaultValue={entry.data.category_id}>
        {(categories.data ?? []).map((category) => (
          <option key={category.id} value={category.id}>{category.name}</option>
        ))}
      </select>
      <select name="person" defaultValue={entry.data.person_id}>
        {(members.data ?? []).map((member) => {
          const profile = Array.isArray(member.profiles) ? member.profiles[0] : member.profiles;
          return <option key={member.user_id} value={member.user_id}>{profile?.display_name}</option>;
        })}
      </select>
      <input type="date" name="date" defaultValue={entry.data.occurred_on} />
      <input name="note" defaultValue={entry.data.note ?? ""} />
      <button type="submit">Sačuvaj</button>
      <button formAction={async () => {
        "use server";
        await deleteEntry(householdId, entryId);
      }}>Obriši</button>
    </form>
  );
}
```
